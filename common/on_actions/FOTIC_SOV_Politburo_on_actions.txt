on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { tag = SOV }
				country_event = { id = FOTIC_SOV_Politburo.1 }
			}
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = {
					tag = SOV
					has_country_flag = fotic_pb_initialized
					NOT = { has_country_flag = fotic_sov_1992_started }
				}
				country_event = { id = FOTIC_SOV_Politburo.100 }
			}
		}
	}
}
