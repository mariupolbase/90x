on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { tag = SOV }
				country_event = { id = FOTIC_Economy.1 }
			}
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = {
					has_country_flag = fotic_economy_active
				}
				country_event = { id = FOTIC_Economy.100 }
			}
		}
	}
}
