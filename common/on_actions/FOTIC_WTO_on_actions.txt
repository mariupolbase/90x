on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = {
					OR = {
						original_tag = POL
						original_tag = DDR
						original_tag = ROM
						original_tag = BUL
						original_tag = HUN
						original_tag = CZE
					}
				}
				add_ideas = idea_wto_membership
				set_rule = {
					can_join_factions = no
					can_create_factions = no
				}
			}
			if = {
				limit = {
					NOT = { has_global_flag = fotic_wto_goals_initialized }
				}
				set_global_flag = fotic_wto_goals_initialized
				every_country = {
					limit = {
						original_tag = SOV
						is_faction_leader = yes
					}
					add_faction_goal_slot = { category = short_term value = 1 }
					add_faction_goal_slot = { category = medium_term value = 1 }
					add_faction_goal_slot = { category = long_term value = 1 }
					add_faction_goal = fotic_wto_short_term_placeholder
					add_faction_goal = fotic_wto_medium_term_placeholder
					add_faction_goal = fotic_wto_long_term_placeholder
				}
			}
		}
	}
	on_monthly = {
		effect = {
			if = {
				limit = {
					has_idea = idea_wto_membership
					NOT = { is_in_faction_with = SOV }
				}
				SOV = { add_to_faction = ROOT }
			}
		}
	}
}
