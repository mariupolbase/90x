country_event = {
	id = FOTIC_SOV_Politburo.100
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				date > 1990.11.30
				NOT = { has_country_flag = fotic_pb_dec90_done }
			}
			set_country_flag = fotic_pb_dec90_done
			set_variable = { fotic_pb_interior_holder = 2 }
			set_variable = { fotic_pb_interior_loyalty = 40 }
			set_variable = { fotic_pb_interior_influence = 55 }
			set_variable = { fotic_pb_vice_holder = 1 }
			set_variable = { fotic_pb_vice_loyalty = 50 }
			set_variable = { fotic_pb_vice_influence = 40 }
		}
		if = {
			limit = {
				has_country_flag = fotic_sov_ev80_fired
				check_variable = { fotic_pb_foreign_holder = 1 }
			}
			set_variable = { fotic_pb_foreign_holder = 2 }
			set_variable = { fotic_pb_foreign_loyalty = 70 }
			set_variable = { fotic_pb_foreign_influence = 35 }
		}
		if = {
			limit = {
				date > 1991.1.13
				NOT = { has_country_flag = fotic_pb_pavlov_done }
			}
			set_country_flag = fotic_pb_pavlov_done
			set_variable = { fotic_pb_premier_holder = 2 }
			set_variable = { fotic_pb_premier_loyalty = 55 }
			set_variable = { fotic_pb_premier_influence = 60 }
		}

		FOTIC_PB_monthly_drift = yes
		FOTIC_PB_clamp_variables = yes
	}

	option = {
		name = FOTIC_SOV_Politburo.100.a
	}
}
